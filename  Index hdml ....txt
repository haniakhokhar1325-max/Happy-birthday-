    <script>
        const interactiveContainer = document.getElementById('interactive-container');
        const topHeading = document.getElementById('top-heading');
        const messageScreen = document.getElementById('message-screen');
        const typingText = document.getElementById('typing-text');

        const message = `Happy birthday my dear wife ❤️

Happppppyyyyy walllllaaaaaa Birthhhhhdaayyyyyyy my sweet wife ...

To aj wo din ha jab ak shehzaaadiiii utri thi asman sy... or wo ha Mera pora jhan, Meri Zindagi, Meri friend, Mera Pyar, Meri muhabat, Mera ishq, Mera sakoon.

Allah Meri pyari si jan je ko lambi Zindagi ata farmaye aur dhear sari khushiyan ata kary, aur har wo cheez jo ap chahti ho. Aur Allah mera ap k sath hamesha kiyamat tak rakhy. Ameen summa Ameen. ❤️

Once again..
Happy birthday!
Happppppyyyyy walllllaaaaaa Happy birthday!

To the Queen Of My Heart, Woman Of My Dreams And The Love Of My Life... 👑

On your birthday I wish that Allah bless you with good luck, happiness, love, good health and success in every walk of your life, you truly deserve it.

Wishing you a very happy birthday sweetheart! You are the love of my life, looking forward to every moment with you. ❤️`;

        function startOpenSequence() {
            interactiveContainer.style.transition = 'all 0.8s ease';
            interactiveContainer.style.transform = 'scale(2)';
            interactiveContainer.style.opacity = '0';

            setTimeout(() => {
                interactiveContainer.style.display = 'none';
                topHeading.style.display = 'none';
                messageScreen.style.opacity = '1';
                messageScreen.style.pointerEvents = 'auto';
                startTyping();
            }, 800);
        }

        let charIndex = 0;
        function startTyping() {
            if (charIndex < message.length) {
                typingText.textContent += message.charAt(charIndex);
                charIndex++;
                setTimeout(startTyping, 40);
            }
        }
